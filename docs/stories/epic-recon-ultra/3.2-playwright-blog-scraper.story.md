### Task 3 — Implement Playwright browser pool

```typescript
class BrowserPool {
  private browser: Browser | null = null;
  private activePagesCount = 0;
  private readonly maxPages = 2;

  async getPage(): Promise<Page> {
    if (this.activePagesCount >= this.maxPages) {
      await this.waitForAvailablePage();
    }
    if (!this.browser) {
      this.browser = await chromium.launch({ headless: true });
    }
    this.activePagesCount++;
    return this.browser.newPage();
  }

  async releasePage(page: Page): Promise<void> {
    await page.close();
    this.activePagesCount--;
  }

  async destroy(): Promise<void> {
    await this.browser?.close();
    this.browser = null;
  }
}
```

### Task 4 — Implement article scraping with Readability

```typescript
import { Readability } from '@mozilla/readability';
import { JSDOM } from 'jsdom';

async function scrapeArticle(
  page: Page,
  url: string
): Promise<BlogScrapedArticle | null> {
  try {
    // Set random UA
    await page.setExtraHTTPHeaders({ 'User-Agent': getRandomUA() });

    // Navigate with timeout
    await page.goto(url, { waitUntil: 'domcontentloaded', timeout: 30000 });

    // Auto-dismiss cookie consent
    await dismissCookieConsent(page);

    // Wait for content hydration
    await page.waitForTimeout(2000);

    // Get page HTML
    const html = await page.content();

    // Primary: Readability.js
    const dom = new JSDOM(html, { url });
    const reader = new Readability(dom.window.document);
    const article = reader.parse();

    if (article && article.textContent.length >= 300) {
      return {
        url,
        title: article.title,
        body_content: article.textContent.slice(0, 5000),
        author: extractAuthor(dom.window.document),
        posted_at: extractPublishedDate(dom.window.document),
        word_count: article.textContent.split(/\s+/).length,
        key_quotes: extractKeyQuotes(dom.window.document),
      };
    }

    // Fallback: CSS selectors (same as url-scraper.ts pattern)
    return await fallbackExtraction(page, url);
  } catch (err) {
    logger.warn({ url, error: (err as Error).message }, '[BLOG-SCRAPER] Article scrape failed');
    return null;
  }
}
```

### Task 5 — Implement rate limiter

```typescript
class DomainRateLimiter {
  private lastRequestTime: Map<string, number> = new Map();
  private readonly minIntervalMs = 20000; // 3 req/min = 20s between requests

  async waitForDomain(domain: string): Promise<void> {
    const lastTime = this.lastRequestTime.get(domain) ?? 0;
    const elapsed = Date.now() - lastTime;
    if (elapsed < this.minIntervalMs) {
      const delay = this.minIntervalMs - elapsed + randomDelay(2000, 5000);
      await new Promise(r => setTimeout(r, delay));
    }
    this.lastRequestTime.set(domain, Date.now());
  }
}
```

  const selectors = [
    '[id*="cookie"] button[id*="accept"]',
    '[class*="cookie"] button[class*="accept"]',
    'button[aria-label*="Accept"]',
    'button[aria-label*="Agree"]',
    '#onetrust-accept-btn-handler',
    '.cookie-banner__accept',

  }

## Dev Notes

### Existing Scraper Reference
`src/modules/manual-news/url-scraper.ts` has a simpler blog scraper using Puppeteer. The new implementation uses Playwright (more stable) + Readability.js (better extraction) + stealth (anti-bot).

### Memory Considerations
Playwright browser instances are heavy (~100-200MB each). The BrowserPool enforces max 2 concurrent pages. The pool's `destroy()` must be called after each recon run to free memory.

### Readability.js
Mozilla's Readability is the same engine Firefox uses for Reader Mode. It strips ads, navigation, and returns clean article text. It requires JSDOM to create a DOM from HTML.
