## Description

As a **ShadowFeed backend**, I need the Recon Ultra service that orchestrates the 5-stage pipeline (profile load → query gen → multi-source fetch → scoring → store) and the controller that exposes HTTP endpoints, so that recon runs can be triggered for any user+pillar combination.

## Tasks

### Task 1 — Implement `recon-ultra.service.ts`

```typescript
export async function runReconUltra(request: ReconUltraRequest): Promise<ReconUltraResult> {
  const startTime = Date.now();
  const { userId, pillarId, sources } = request;
  const pillarConfig = getPillarConfig(pillarId);

  // Stage 0: Load profile
  const userProfile = await getUserProfile(userId);
  const profileKeywords = extractProfileKeywords(...);

  // Stage 1: Generate queries
  const queries = await generateReconQueries(userId, pillarId);

  // Stage 2: Multi-source fetch (parallel)
  const sourcesToRun = sources ?? ['google_news', 'google_trends', 'reddit', 'twitter', 'blog'];
  const fetchResults = await Promise.allSettled(
    sourcesToRun.map(source => fetchBySource(source, queries, userId, profileKeywords))
  );

  // Merge + filter + dedup
  const allCandidates = mergeResults(fetchResults);
  const filtered = applyEditorialFilter(allCandidates);
  const deduped = deduplicateByTitle(filtered);

  // Stage 3: Score
  const scored = scoreAllReconCandidates(deduped, profileKeywords, pillarConfig);

  // Stage 4: Store
  await storeIntelItems(userId, pillarId, scored);

  return { userId, pillarId, candidates: scored, ... };
}
```

### Task 2 — Implement source dispatcher

```typescript
async function fetchBySource(
  source: ReconSourceType,
  queries: string[],
  userId: string,
  profileKeywords: string[]
): Promise<ReconUltraCandidate[]> {
  switch (source) {
    case 'google_news': return fetchNewsDynamic(queries, profileKeywords);
    case 'google_trends': return fetchTrendsDynamic(profileKeywords);
    case 'reddit': return fetchRedditDynamic(queries, userId);
    case 'twitter': return fetchTwitterDynamic(queries);
    case 'blog': return fetchBlogArticles(userId, queries, profileKeywords);
  }
}
```

- Service pattern: `src/modules/recon/recon.service.ts`
- Controller pattern: `src/modules/recon/recon.controller.ts`
