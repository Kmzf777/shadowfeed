## Tasks

### Task 1 — Implement `fetchNewsDynamic()`

```typescript
import RSSParser from 'rss-parser';
import type { ReconUltraCandidate } from '../recon-ultra.types.js';

const rssParser = new RSSParser();

export async function fetchNewsDynamic(
  queries: string[],
  profileKeywords: string[]
): Promise<ReconUltraCandidate[]> {
  const fetchPromises = queries.map(q => fetchNewsForQuery(q, profileKeywords));
  const results = await Promise.allSettled(fetchPromises);

  const candidates: ReconUltraCandidate[] = [];
  for (const r of results) {
    if (r.status === 'fulfilled') candidates.push(...r.value);
  }
  return candidates;
}

async function fetchNewsForQuery(query: string, keywords: string[]): Promise<ReconUltraCandidate[]> {
  const encoded = encodeURIComponent(query);
  const feedUrl = `https://news.google.com/rss/search?q=${encoded}&hl=en-US&gl=US&ceid=US:en`;
  const feed = await rssParser.parseURL(feedUrl);

  return feed.items.slice(0, 8)
    .filter(item => (item.title ?? '').length >= 5)
    .map(item => ({
      source_type: 'google_news' as const,
      title: item.title ?? '',
      summary: item.contentSnippet?.slice(0, 500) ?? null,
      body_content: null,
      url: item.link ?? null,
      author: item.creator ?? null,
      category: null,
      source_score: null,
      source_comments: null,
      source_retweets: null,
      source_views: null,
      recency_score: 0,       // computed by profile-scorer
      engagement_score: 0,     // computed by profile-scorer
      relevance_score: 0,      // computed by profile-scorer
      richness_score: 0,       // computed by profile-scorer
      pillar_alignment_score: 0,
      final_score: 0,
      pillar_tag: 'educate' as const,  // default, overridden by scorer
      query_used: query,
      posted_at: item.isoDate ?? null,
      expires_at: '',          // set by service
    }));
}
```
